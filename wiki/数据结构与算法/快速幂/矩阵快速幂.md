矩阵乘法，矩阵A：m x k，矩阵B: k x n，矩阵A x B: m x n
只要实现矩阵乘法，就能实现
```cpp
using ll = long long;
using matrix = vector<vector<ll>>;
static constexpr int mod = 1e9 + 7;
matrix mul(const matrix& A, const matrix& B){
    int n = A.size();
    int t = A[0].size();
    int m = B[0].size();
    matrix Res = matrix(n, vector<ll>(m, 0));
    for(int i = 0; i < n; ++i){
        for(int j = 0; j < m; ++j){
            for(int k = 0; k < t; ++k){
                Res[i][j] = (Res[i][j] + A[i][k] * B[k][j]) % mod;
            }
        }
    }
    return Res;
}
matrix quickPower(matrix M, ll k){
    int a = M.size();
    matrix Res(a, vector<ll>(a, 0));
    for(int i = 0; i < a; ++i){
        Res[i][i] = 1;
    }
    while(k){
        if(k & 1){
            Res = mul(Res, M);
        }
        M = mul(M, M);
        k >>= 1;
    }
    return Res;
}
```

## 矩阵快速幂与动态规划
### 一维二阶问题
### 斐波那契数列 矩阵快速幂形式
递推公式：
$$F_n=F_{n-1}+F_{n-2}$$
初值：
$$F_0=0,\quad F_1=1$$

### 1. 转换成矩阵形式
目标：
把
$$\begin{bmatrix} F_n\\ F_{n-1} \end{bmatrix}$$
转移到
$$\begin{bmatrix} F_{n+1}\\ F_n \end{bmatrix}$$

根据递推：
$$F_{n+1}=F_n+F_{n-1}$$

所以：
$$
\begin{bmatrix} F_{n+1}\\ F_n \end{bmatrix}
=
\begin{bmatrix} 1&1\\ 1&0 \end{bmatrix}
\begin{bmatrix} F_n\\ F_{n-1} \end{bmatrix}
$$

记矩阵
$$A= \begin{bmatrix} 1&1\\ 1&0 \end{bmatrix}$$

递推展开：
$$
\begin{bmatrix} F_n\\ F_{n-1} \end{bmatrix}
= A^{n-1}
\begin{bmatrix} F_1\\ F_0 \end{bmatrix}
$$

代入初值 $F_1=1,\ F_0=0$：
$$
\begin{bmatrix} F_n\\ F_{n-1} \end{bmatrix}
=
\begin{bmatrix} 1&1\\ 1&0 \end{bmatrix}^{n-1}
\begin{bmatrix} 1\\ 0 \end{bmatrix}
$$
链接:https://leetcode.cn/problems/fibonacci-number/description/
求第n个斐波那契数
```cpp
class Solution {
public:
    using matrix = vector<vector<int>>;
    matrix mul(const matrix& a, const matrix& b){
        int n = a.size();
        int r = a[0].size();
        int m = b[0].size();
        matrix res(n, vector<int>(m, 0));
        for(int i = 0; i < n; ++i){
            for(int j = 0; j < m; ++j){
                for(int k = 0; k < r; ++k){
                    res[i][j] += a[i][k] * b[k][j];
                }
            }
        }
        return res;
    }
    matrix qpow(matrix a, long long k){
        int n = a.size();
        matrix res(n, vector<int>(n, 0));
        for(int i = 0; i < n; ++i){
            res[i][i] = 1;
        }
        while(k){
            if(k & 1){
                res = mul(res, a);
            }
            a = mul(a, a);
            k >>= 1;
        }
        return res;
    }
    int fib(int n) {
        if(n == 0){
            return 0;
        }
        // [1, 0]
        vector<vector<int>> f1(1, {1,  0});
        vector<vector<int>> a = {
            {1, 1},
            {1, 0}
        };
        matrix res = mul(f1, qpow(a, n - 1));
        return res[0][0];
    }
};
```
